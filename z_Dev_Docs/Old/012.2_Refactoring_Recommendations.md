import random
from abc import ABC, abstractmethod
from typing import List, Dict, Optional, Any
from collections import Counter

import pandas as pd


class LottoAlgorithm(ABC):
    """
    로또 알고리즘 추상 베이스 클래스
    
    모든 알고리즘은 이 클래스를 상속받아 구현
    
    2026-01-08 05:20:00 EST - 공통 헬퍼 메서드 추가:
    - calculate_frequency: 빈도 계산
    - generate_random_sets: 랜덤 번호 생성 (fallback)
    - apply_temperature: 온도 파라미터 적용
    - apply_recent_draw_penalty: 직전 회차 페널티
    """
    
    def __init__(
        self,
        algorithm_id: int,
        name: str,
        description: str,
        cost_per_set: int = 0
    ):
        """(기존 코드 유지)"""
        self.algorithm_id = algorithm_id
        self.name = name
        self.description = description
        self.cost_per_set = cost_per_set
        self.version = "1.0"
    
    # ========================================

    # ========================================
    
    @abstractmethod
    def generate_numbers(...):
        """(기존 코드)"""

    def validate_parameters(...):
        """(기존 코드)"""
        pass

        """(기존 코드)"""

        """(기존 코드)"""

    def calculate_frequency(
        self,
        data: pd.DataFrame,
        include_zero_freq: bool = True
    ) -> Dict[int, int]:
        """
        번호별 출현 빈도 계산 (공통 헬퍼 메서드)
        
        알고리즘 2, 4, 6에서 중복 제거
        
        Args:
            data: 과거 당첨번호 DataFrame
            include_zero_freq: 0회 출현 번호도 포함 여부 (기본 True)
        
        Returns:
            Dict[int, int]: {번호: 출현 횟수}
        
        Example:
            >>> frequency = self.calculate_frequency(historical_data)
            >>> print(frequency[12])  # 12번 출현 횟수
            45
        """
        frequency = Counter()
        
        for _, row in data.iterrows():
            for i in range(1, 7):
                frequency[row[f'num{i}']] += 1
        
        if include_zero_freq:
            for num in range(1, 46):
                if num not in frequency:
                    frequency[num] = 0
        
        return dict(frequency)
    
    def generate_random_sets(
        self,
        n_sets: int,
        exclude_numbers: Optional[List[int]] = None,
        include_numbers: Optional[List[int]] = None
    ) -> List[List[int]]:
        """
        순수 랜덤 번호 생성 (공통 fallback 메서드)
        
        알고리즘 2, 5, 6, 7에서 중복 제거
        과거 데이터 없을 때 모든 알고리즘이 사용
        
        Args:
            n_sets: 생성할 세트 수
            exclude_numbers: 제외할 번호 리스트
            include_numbers: 반드시 포함할 번호 리스트
        
        Returns:
            List[List[int]]: 생성된 번호 세트 리스트
        
        Example:
            >>> if historical_data is None:
            ...     return self.generate_random_sets(5, exclude, include)
        """
        results = []
        available_numbers = self.get_available_numbers(
            exclude_numbers, include_numbers
        )
        
        for _ in range(n_sets):
            numbers = []
            
            if include_numbers:
                numbers.extend(include_numbers)
            
            remaining_count = 6 - len(numbers)
            selected = random.sample(available_numbers, remaining_count)
            numbers.extend(selected)
            
            results.append(sorted(numbers))
        
        return results
    
    @staticmethod
    def apply_temperature(
        probabilities: Dict[int, float],
        temperature: float
    ) -> Dict[int, float]:
        """
        확률 분포에 온도 파라미터 적용 (공통 유틸리티)
        
        알고리즘 2, 3에서 중복 제거
        
        Args:
            probabilities: 원본 확률 분포 {번호: 확률}
            temperature: 온도 파라미터 (0.1~3.0)
                - < 1.0: 확률 차이 극대화 (날카로운 분포)
                - = 1.0: 원본 유지 (변경 없음)
                - > 1.0: 확률 차이 완화 (평탄한 분포)
        
        Returns:
            Dict[int, float]: 조정된 확률 분포
        
        Example:
            >>> probs = {1: 0.1, 2: 0.2, 3: 0.7}
            >>> adjusted = LottoAlgorithm.apply_temperature(probs, 0.5)
            >>> # 0.7 확률이 더 강조됨
        
        References:
            - Softmax temperature scaling
            - 010.1_Algorithm_Redesign.md 섹션 D
        """
        if temperature == 1.0:
            return probabilities
        
        # P^(1/T) 변환
        adjusted = {
            num: p ** (1 / temperature) 
            for num, p in probabilities.items()
        }
        
        # 정규화 (합계를 1로)
        total = sum(adjusted.values())
        if total > 0:
            return {num: adj / total for num, adj in adjusted.items()}
        
        return probabilities
    
    @staticmethod
    def apply_recent_draw_penalty(
        probabilities: Dict[int, float],
        historical_data: pd.DataFrame,
        penalty_rate: float
    ) -> Dict[int, float]:
        """
        직전 회차 출현 번호 확률 할인 (공통 유틸리티)
        
        알고리즘 2, 3에서 중복 제거
        
        Args:
            probabilities: 원본 확률 분포 {번호: 확률}
            historical_data: 과거 당첨번호 DataFrame
            penalty_rate: 할인율 (0.1~1.0)
                - 0.5: 확률을 절반으로 (50% 할인)
                - 0.33: 확률을 1/3로 (66% 할인)
        
        Returns:
            Dict[int, float]: 조정된 확률 분포
        
        Example:
            >>> # 직전 회차: [5, 12, 23, 31, 38, 42]
            >>> probs = {5: 0.02, 12: 0.022, ...}
            >>> adjusted = LottoAlgorithm.apply_recent_draw_penalty(
            ...     probs, historical_data, penalty_rate=0.5
            ... )
            >>> # 5번, 12번 등의 확률이 절반으로 감소
        
        References:
            - 010.1_Algorithm_Redesign.md 섹션 B-3
        """
        if len(historical_data) < 1:
            return probabilities
        
        # 직전 회차 번호 추출
        latest_draw = historical_data.tail(1).iloc[0]
        latest_numbers = {latest_draw[f'num{i}'] for i in range(1, 7)}
        
        # 확률 할인 적용
        adjusted = probabilities.copy()
        for num in latest_numbers:
            if num in adjusted:
                adjusted[num] *= penalty_rate
        
        # 정규화 (합계를 1로)
        total = sum(adjusted.values())
        if total > 0:
            adjusted = {num: prob / total for num, prob in adjusted.items()}
        
        return adjusted

import pytest
import pandas as pd
from app.algorithms.base import LottoAlgorithm

class TestCalculateFrequency:
    """calculate_frequency 메서드 테스트"""
    
    def test_basic_frequency(self):
        """기본 빈도 계산"""
        algo = AdvancedFrequencyAlgorithm()
        
        data = pd.DataFrame({
            'num1': [1, 1, 2],
            'num2': [2, 2, 3],
            'num3': [3, 3, 4],
            'num4': [4, 4, 5],
            'num5': [5, 5, 6],
            'num6': [6, 6, 7],
        })
        
        frequency = algo.calculate_frequency(data)
        
        assert frequency[1] == 2  # 1번이 2회
        assert frequency[2] == 3  # 2번이 3회
        assert frequency[8] == 0  # 8번은 0회
    
    def test_zero_frequency_included(self):
        """0회 출현 번호 포함"""
        algo = AdvancedFrequencyAlgorithm()
        
        data = pd.DataFrame({
            'num1': [1],
            'num2': [2],
            'num3': [3],
            'num4': [4],
            'num5': [5],
            'num6': [6],
        })
        
        frequency = algo.calculate_frequency(data, include_zero_freq=True)
        
        assert len(frequency) == 45
        assert frequency[45] == 0


class TestGenerateRandomSets:
    """generate_random_sets 메서드 테스트"""
    
    def test_basic_generation(self):
        """기본 랜덤 생성"""
        algo = AdvancedFrequencyAlgorithm()
        
        results = algo.generate_random_sets(5)
        
        assert len(results) == 5
        assert all(len(nums) == 6 for nums in results)
        assert all(all(1 <= n <= 45 for n in nums) for nums in results)
    
    def test_with_exclude_include(self):
        """제외/포함 번호"""
        algo = AdvancedFrequencyAlgorithm()
        
        results = algo.generate_random_sets(
            3,
            exclude_numbers=[1, 2, 3],
            include_numbers=[44, 45]
        )
        
        for nums in results:
            assert 44 in nums
            assert 45 in nums
            assert 1 not in nums
            assert 2 not in nums
            assert 3 not in nums


class TestApplyTemperature:
    """apply_temperature 정적 메서드 테스트"""
    
    def test_temperature_1(self):
        """온도 1.0 (변경 없음)"""
        probs = {1: 0.1, 2: 0.2, 3: 0.7}
        
        result = LottoAlgorithm.apply_temperature(probs, 1.0)
        
        assert result == probs
    
    def test_temperature_low(self):
        """온도 < 1.0 (날카로운 분포)"""
        probs = {1: 0.1, 2: 0.2, 3: 0.7}
        
        result = LottoAlgorithm.apply_temperature(probs, 0.5)
        
        # 0.7이 더 강조되어야 함
        assert result[3] > probs[3]
        assert sum(result.values()) == pytest.approx(1.0)

    """apply_recent_draw_penalty 정적 메서드 테스트"""

        """기본 패널티 적용"""
        data = pd.DataFrame({

            'num2': [12],
            'num3': [23],
            'num4': [31],
            'num5': [38],
            'num6': [42],
        })
        
        probs = {n: 1/45 for n in range(1, 46)}
        
        result = LottoAlgorithm.apply_recent_draw_penalty(

        assert result[5] < probs[5]

        # 다른 번호는 상대적으로 증가
        assert result[1] > probs[1]
        
        # 합계는 여전히 1
